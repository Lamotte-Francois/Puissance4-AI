class ReplaceBlackWithBackground:
    def __init__(self, background_path):
        bg = cv2.imread(background_path)
        bg = cv2.cvtColor(bg, cv2.COLOR_BGR2RGB)
        self.background = bg.astype(np.float32) / 255.0

    def __call__(self, img):
        img_np = img.permute(1, 2, 0).numpy()

        mask = np.all(img_np < 0.01, axis=-1)

        bg = cv2.resize(self.background, (img_np.shape[1], img_np.shape[0]))

        img_np[mask] = bg[mask]

        return torch.tensor(img_np, dtype=torch.float32).permute(2, 0, 1)

class  CustomDataset ( Dataset ): 
    def  __init__ ( self, path, transforms):

        """ 
        Ces étiquettes et données peuvent être le chemin d'accès au jeu de données ou directement. 
        Dans cet exemple, je définis un petit jeu de données. 
        Cependant, selon le contexte, le code peut être modifié pour lire les données et les étiquettes 
        à partir de leur emplacement. 


        Les transformations sont initialisées ici et seront appliquées aux données et aux étiquettes 
        lors de leur récupération dans la méthode __getitem__(). Ici, la transformation peut être 
        n'importe quelle fonction selon la tâche à accomplir. 
        
        Arguments : 
            data (liste ou tableau) : Échantillons de données d'entrée. 
            labels (liste ou tableau) : Étiquettes correspondantes aux échantillons de données. 
            transform (fonction, optionnel) : Transformations pour le prétraitement des données. 
            target_transform (fonction, optionnel) : Transformations pour le prétraitement des étiquettes. 
        """

        #self.data = data 

        self.data = load_dataset(path)
        

        self.horizontalflip = v2.RandomHorizontalFlip(1)
        self.transform = transforms

    def  __len__ ( self ): 
        return  len(self.data) 
    

    def __getitem__(self, idx):
        bb1, bb2, turn, probs = self.data[idx]
        grid = bitboard_to_grid(bb1, bb2)
        img = draw_board_cv2(grid, probs, turn)
        img = torch.tensor(img, dtype=torch.float32)

#
        img = img.permute(2, 0, 1)
#

        label = torch.tensor(probs, dtype=torch.float32)

        # -----------------

        img = self.transform(img)


        # ------------------
        
        img,label = self.random_horizontal_flip(img,label)

        return img, label

    
    def random_horizontal_flip(self, grid, probs, p=0.5):

        if random.random() < p:
            grid = self.horizontalflip(grid)
            probs = self.horizontalflip(probs)
            
        return   grid,probs
