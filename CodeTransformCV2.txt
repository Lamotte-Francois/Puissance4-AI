import torch
import random
import torchvision.transforms.v2 as v2
from torch.utils.data import Dataset
from grille import load_dataset, bitboard_to_grid, draw_board_cv2
import cv2
import numpy as np
import os


# ----------------------------
# BACKGROUND DATASET
# ----------------------------
class BackgroundDataset:
    def __init__(self, root_path):
        self.images = []

        for root, _, files in os.walk(root_path):
            for f in files:
                if f.lower().endswith((".jpg", ".jpeg", ".png")):
                    # DEBUG FLEXIBLE: accepte kitchen / dining_room partout dans le chemin
                    path_lower = root.lower()

                    if ("kitchen" in path_lower) or ("dining_room" in path_lower) or ("computerroom" in path_lower) or ("livingroom" in path_lower) or ("classroom" in path_lower):
                        self.images.append(os.path.join(root, f))

        print("Background images loaded:", len(self.images))

        if len(self.images) == 0:
            raise ValueError("Aucune image kitchen/dining_room trouvée -> vérifie le chemin du dataset")

    def get_random(self):
        path = random.choice(self.images)

        img = cv2.imread(path)
        if img is None:
            return self.get_random()

        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        img = img.astype(np.float32) / 255.0

        return img


# ----------------------------
# TRANSFORM: BACKGROUND REPLACEMENT
# ----------------------------
class ReplaceBlackWithBackground:
    def __init__(self, background_dataset):
        self.bg_dataset = background_dataset

    def __call__(self, img):
        img_np = img.permute(1, 2, 0).numpy()

        mask = np.all(img_np < 0.01, axis=-1)

        bg = self.bg_dataset.get_random()
        bg = cv2.resize(bg, (img_np.shape[1], img_np.shape[0]))

        img_np[mask] = bg[mask]

        return torch.tensor(img_np, dtype=torch.float32).permute(2, 0, 1)


# ----------------------------
# DATASET
# ----------------------------
class CustomDataset(Dataset):
    def __init__(self, path, transforms):

        self.data = load_dataset(path)

        self.horizontalflip = v2.RandomHorizontalFlip(1)
        self.transform = transforms

    def __len__(self):
        return len(self.data)

    def __getitem__(self, idx):
        bb1, bb2, turn, probs = self.data[idx]

        grid = bitboard_to_grid(bb1, bb2)
        img = draw_board_cv2(grid, probs, turn)

        img = torch.tensor(img, dtype=torch.float32).permute(2, 0, 1)
        label = torch.tensor(probs, dtype=torch.float32)

        img = self.transform(img)
        img, label = self.random_horizontal_flip(img, label)

        return img, label

    def random_horizontal_flip(self, grid, probs, p=0.5):
        if random.random() < p:
            grid = self.horizontalflip(grid)
            probs = self.horizontalflip(probs)
        return grid, probs
