import cv2
import torch
import torch.nn as nn
import torchvision.models as models
import torchvision.transforms.v2 as v2
import torch.nn.functional as F

# =========================================================
# MODEL (RESNET50 COMPATIBLE CHECKPOINT)
# =========================================================

class Connect4Model(nn.Module):
    def __init__(self):
        super().__init__()

        # Backbone ResNet50 (comme dans ton training)
        self.backbone = models.resnet50(weights=None)
        self.backbone.fc = nn.Identity()  # sortie = 2048

        # Heads
        self.policy = nn.Linear(2048, 7)
        self.player = nn.Linear(2048, 2)

    def forward(self, x):
        x = self.backbone(x)
        policy = self.policy(x)
        player = self.player(x)
        return policy, player


# =========================================================
# DEVICE
# =========================================================

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Device:", device)

# =========================================================
# LOAD MODEL
# =========================================================

model = Connect4Model().to(device)

checkpoint = torch.load(
    r"D:\\Master_2025-2026\\Systèmes_intelligents\\AllCode\\test\\best.pth",
    map_location=device
)

model.load_state_dict(checkpoint)
model.eval()

# =========================================================
# TRANSFORMS
# =========================================================

transform = v2.Compose([
    v2.Resize((224, 224))
])

# =========================================================
# CAMERA
# =========================================================

cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()
    if not ret:
        break

    # -------------------------------------------------
    # PREPROCESS
    # -------------------------------------------------

    img = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

    img = torch.from_numpy(img).float() / 255.0
    img = img.permute(2, 0, 1)

    img = transform(img)

    img = img.unsqueeze(0).to(device)

    # =================================================
    # INFERENCE
    # =================================================

    with torch.no_grad():
        policy, player = model(img)

        probs = F.softmax(policy, dim=1)

        best_move = probs.argmax(1).item()
        confidence = probs.max().item()

        current_player = player.argmax(1).item()

    # =================================================
    # DISPLAY
    # =================================================

    text = f"Move: {best_move} | Conf: {confidence:.2f}"

    cv2.putText(
        frame,
        text,
        (30, 50),
        cv2.FONT_HERSHEY_SIMPLEX,
        1,
        (0, 255, 0),
        2
    )

    player_text = f"Player: {current_player}"

    cv2.putText(
        frame,
        player_text,
        (30, 100),
        cv2.FONT_HERSHEY_SIMPLEX,
        1,
        (255, 0, 0),
        2
    )

    cv2.imshow("Connect4 AI", frame)

    if cv2.waitKey(1) & 0xFF == ord("q"):
        break

cap.release()
cv2.destroyAllWindows()
