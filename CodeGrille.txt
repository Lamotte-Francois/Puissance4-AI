COLS = 7
ROWS = 6
CELL_SIZE = 100
#EXTRA_HEIGHT = 80  # espace pour les probas

# Couleurs (BGR pour OpenCV)
BLUE = (255, 0, 0)
BLACK = (0, 0, 0)
RED = (0, 0, 255)
YELLOW = (0, 255, 255)
WHITE = (255, 255, 255)


def bitboard_to_grid(bb1, bb2):
    grid = [[0 for _ in range(COLS)] for _ in range(ROWS)]

    for col in range(COLS):
        for row in range(ROWS):
            bit_index = col * 7 + row
            mask = 1 << bit_index
            display_row = ROWS - 1 - row

            if bb1 & mask:
                grid[display_row][col] = 1
            elif bb2 & mask:
                grid[display_row][col] = 2

    return grid

def load_dataset(path):
    try:
        with open(path, "r") as f:
            raw_data = json.load(f)

        dataset = []

        for key, value in raw_data.items():
            key = key.strip("()")
            bb1_str, bb2_str = key.split(",")

            bb1 = int(bb1_str.strip())
            bb2 = int(bb2_str.strip())

            probs = value[0]
            turn = value[1]

            dataset.append((bb1, bb2, turn, probs))

        return dataset

    except Exception as e:
        print(f"Erreur chargement dataset : {e}")
        sys.exit()


def main():
    dataset = load_dataset("E:\\dataset.json")

    index = 0
    bb1, bb2, turn, probs = dataset[index]
    grid = bitboard_to_grid(bb1, bb2)

    delay = 500  # ms
    last_update = cv2.getTickCount()

    while True:
        now = cv2.getTickCount()
        elapsed = (now - last_update) / cv2.getTickFrequency() * 1000

        if elapsed >= delay:
            index = (index + 1) % len(dataset)
            bb1, bb2, turn, probs = dataset[index]
            grid = bitboard_to_grid(bb1, bb2)
            last_update = now

        img = draw_board_cv2(grid, probs, turn)

        cv2.imshow("Puissance 4 - Dataset Viewer", img)
        # ESC pour quitter
        if cv2.waitKey(1) & 0xFF == 27:
            break

    cv2.destroyAllWindows()


if __name__ == "__main__":
    main()
